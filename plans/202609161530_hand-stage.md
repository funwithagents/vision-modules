## Target API (exact names)

```python
@dataclass(frozen=True)
class DetectedHand:
    box: tuple[float, float, float, float]   # (x0, y0, x1, y1) normalized to [0, 1], UNPADDED
    score: float

class HandDetector(Protocol):
    def detect(self, image_bgr: np.ndarray, ts: float) -> tuple[DetectedHand, ...]: ...
    def close(self) -> None: ...

@dataclass(frozen=True, eq=False)
class Hand:
    bbox: tuple[int, int, int, int]   # pixels, padded + clamped
    crop: np.ndarray | None           # independent copy; None only for a zero-area box
    score: float

@dataclass(frozen=True, eq=False)
class HandResult(Result):
    hands: tuple[Hand, ...]
    @property
    def first(self) -> Hand | None: ...

def padded_box(box: tuple[float, float, float, float], width: int, height: int, pad: float) -> tuple[int, int, int, int]:
    """Normalized unpadded box -> padded, clamped pixel box. Pure function."""

class HandStage(Stage[Frame, HandResult]):
    def __init__(self, provider: Upstream[Frame], target_fps: float | None = 30, pad: float = 0.35,
```

### `padded_box` (write it exactly like this)

```
x0, y0, x1, y1 = box[0]*width, box[1]*height, box[2]*width, box[3]*height
bw, bh = x1 - x0, y1 - y0
x0 -= pad*bw; x1 += pad*bw; y0 -= pad*bh; y1 += pad*bh
return (int(max(0, x0)), int(max(0, y0)), int(min(width, x1)), int(min(height, y1)))
```

### `HandStage.process`

1. If `self._detector is None`: `self._detector = self._given or MediaPipeHandDetector(num_hands=self.max_hands)` (lazy, so it is created on the worker thread).
2. `found = self._detector.detect(frame.image, frame.ts)`; sort by `score` descending; keep the first `max_hands`.
3. For each: `bbox = padded_box(d.box, w, h, pad)`; `crop = frame.image[y0:y1, x0:x1].copy()` if `x1 > x0 and y1 > y0` else `None`; `Hand(bbox, crop, d.score)`.
4. Return `HandResult(frame.frame_id, frame.ts, present=bool(hands), hands=tuple(hands))`.

### `MediaPipeHandDetector` (Tasks API, `VIDEO` mode)

```python
import mediapipe as mp
from mediapipe.tasks.python import BaseOptions, vision

options = vision.HandLandmarkerOptions(
    base_options=BaseOptions(model_asset_path=str(path)),
    running_mode=vision.RunningMode.VIDEO,
    num_hands=num_hands,
    min_hand_detection_confidence=...,
    min_hand_presence_confidence=...,
    min_tracking_confidence=...,
)
self._lm = vision.HandLandmarker.create_from_options(options)
```

`detect()`: `rgb = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2RGB)`; `mp_img = mp.Image(image_format=mp.ImageFormat.SRGB, data=rgb)`; `ts_ms = int(ts * 1000)`; if `ts_ms <= self._last_ts_ms`: `ts_ms = self._last_ts_ms + 1` (MediaPipe requires strictly increasing timestamps); `res = self._lm.detect_for_video(mp_img, ts_ms)`; for each `i`: `xs = [p.x for p in res.hand_landmarks[i]]`, same for `ys`; `box = (min(xs), min(ys), max(xs), max(ys))` clamped to `[0, 1]`; `score = res.handedness[i][0].score`. `close()`: `self._lm.close()`.

`ensure_model`: `path.parent.mkdir(parents=True, exist_ok=True)`; if `path.exists()` return it; else `urllib.request.urlretrieve(url, tmp)` with `tmp = path.with_suffix(".part")`, then `tmp.replace(path)`. Log one line before downloading ("downloading hand landmarker model to …").

## Live test (`tests-e2e/test_hand_live.py`)

```python
def test_mediapipe_detector_runs_on_a_blank_image():
    require_env("VISION_MODULES_E2E")  # skip unless opted in (network: model download)
    det = MediaPipeHandDetector()
    try:
        blank = np.zeros((480, 640, 3), np.uint8)
        out = det.detect(blank, 0.0)
        assert out == ()  # no hand in a black image
        out2 = det.detect(blank, 0.0)  # same ts twice must not raise (timestamp bump)
        assert out2 == ()
    finally:
        det.close()


def test_mediapipe_detector_finds_a_hand_in_a_real_photo():
    require_env("VISION_MODULES_E2E")
    path = require_env(
        "VISION_MODULES_HAND_IMAGE"
    )  # path to any photo with one clearly visible hand
    img = cv2.imread(path)
    assert img is not None
    det = MediaPipeHandDetector()
    try:
        out = det.detect(img, 0.0)
        assert len(out) >= 1
        x0, y0, x1, y1 = out[0].box
        assert 0 <= x0 < x1 <= 1 and 0 <= y0 < y1 <= 1
    finally:
        det.close()
```
